\title{A Counterexample to TLMC Conjecture 00000000548:\\
$\iota(I)\le\beta_0(I)$ Fails for Principal Monomial Ideals}
\author{TLMC falsification package}
\date{October 2, 2026}

\begin{document}
\maketitle

\begin{abstract}
TLMC conjecture 00000000548 asserts that for a monomial ideal $I$, the minimal
number $\iota(I)$ of components in an irredundant irreducible decomposition is
bounded by $\beta_0(I)$, the minimal number of monomial generators, claiming
the bound is combinatorially verified for monomial ideals up to $n=12$ with
tightness attained by complete intersections. We exhibit the principal monomial
ideal $(xyz)\subseteq k[x,y,z]$, for which
\[
(xyz)=(x)\cap(y)\cap(z),\qquad \iota\big((xyz)\big)=3>1=\beta_0\big((xyz)\big),
\]
refuting the conjecture at $n=3$ --- inside the claimed verification range ---
by an ideal that is itself a complete intersection. All numbers are
independently recomputed by exhaustive enumeration (\texttt{reproduce.py}) and
formalized in axiom-free Lean 4 (\texttt{lean4/}).
\end{abstract}

\section{The conjecture}

\begin{quote}
\textbf{00000000548.} \emph{Definition: the minimal number $\iota(I)$ of
irreducible decompositions is the least number of components. Conjecture:
$\iota(I)\le\beta_0(I)$; and the combinatorial control of the decomposition
number is verified up to $n=12$ for monomial ideals, with tightness attained
by complete intersections.}
\end{quote}

Throughout, $R=k[x_1,\dots,x_n]$ and all ideals are monomial ideals, identified
with their sets of exponent vectors $e\in\mathbb{N}^n$.

\begin{definition}\label{def:iota}
An \emph{irreducible} monomial ideal is one generated by pure powers of a
subset of the variables, $(x_{i_1}^{a_1},\dots,x_{i_s}^{a_s})$. An
\emph{irreducible decomposition} of $I$ is an expression $I=\bigcap_j J_j$
with all $J_j$ irreducible; it is \emph{irredundant} if no component can be
dropped. $\iota(I)$ is the least number of components of an irredundant
irreducible decomposition of $I$.
\end{definition}

\begin{definition}\label{def:beta0}
$\beta_0(I)$ is the cardinality of the unique minimal monomial generating set
of $I$ (the number of generators of $I$, i.e.\ the rank of
$\operatorname{Tor}_0(k,I)$).
\end{definition}

\section{The counterexample}

Work in $R=k[x,y,z]$ and let $I=(xyz)$. Write $e=(e_x,e_y,e_z)$ for exponent
vectors, and let $|_1,|_2,|_3$ denote the coordinates. The ideal generated by a
monomial $x^a y^b z^c$ is the up-set $\{e : e_x\ge a,\ e_y\ge b,\ e_z\ge c\}$;
the irreducible ideal $(x^a,y^b,z^c)$ is
$\{e : e_x\ge a \text{ or } e_y\ge b \text{ or } e_z\ge c\}$ (absent variables
carry no constraint). Both descriptions are componentwise, so all equalities
and inequalities of monomial sets verified inside a finite exponent box lift
verbatim to $R$.

\begin{theorem}\label{thm:decomposition}
$(xyz)=(x)\cap(y)\cap(z)$.
\end{theorem}

\begin{proof}
$xyz$ divides a monomial $x^{e_x}y^{e_y}z^{e_z}$ iff $e_x\ge1$ and $e_y\ge1$
and $e_z\ge1$, iff the monomial lies in each of $(x)$, $(y)$, $(z)$.
\end{proof}

\begin{lemma}\label{lem:irreducible}
$(x)$, $(y)$, $(z)$ are irreducible monomial ideals. In fact, no two monomial
ideals strictly containing $(x)$ intersect to $(x)$, and similarly for $(y)$
and $(z)$.
\end{lemma}

\begin{proof}
Suppose $P$ and $Q$ are monomial ideals with $P\supsetneq(x)$ and
$Q\supsetneq(x)$. Choose $u\in P$, $v\in Q$ with $u_x=v_x=0$ (such monomials
exist because $P,Q$ strictly contain $(x)$, which consists exactly of the
monomials with $x$-exponent $\ge1$). Then $w:=uv\in P$ and $w\in Q$ since both
are up-sets, while $w_x=u_x+v_x=0<1$, so $w\notin(x)$. Hence
$P\cap Q\supsetneq(x)$. The cases of $(y)$ and $(z)$ are symmetric.
\end{proof}

\begin{lemma}\label{lem:irredundant}
The decomposition of Theorem~\ref{thm:decomposition} is irredundant.
\end{lemma}

\begin{proof}
$yz\in(y)\cap(z)$ but $yz\notin(xyz)$, so dropping $(x)$ strictly enlarges the
intersection; $xz$ and $xy$ play the same role for $(y)$ and $(z)$.
\end{proof}

\begin{theorem}\label{thm:iota}
$\iota\big((xyz)\big)=3$.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:decomposition} and Lemmas~\ref{lem:irreducible} and
\ref{lem:irredundant} there is an irredundant irreducible decomposition with
$3$ components, so $\iota\le3$. Conversely, irreducible monomial ideals are
exactly the pure-power-generated ones, i.e.\ the ideals
$J(a,b,c)=\{e: e_x\ge a \text{ or } e_y\ge b \text{ or } e_z\ge c\}$ with
$(a,b,c)\in(\{0\}\cup\mathbb{N}_{\ge1})^3$, $0$ meaning the variable is
absent. An exhaustive enumeration of all $6^3=216$ such ideals inside the box
$[0,4]^3$ (script \texttt{reproduce.py}, checks~5b--5d) shows that no single
$J(a,b,c)$ equals $(xyz)\cap[0,4]^3$ and no intersection $J_1\cap J_2$ of two
of them does, whereas $J(1,0,0)\cap J(0,1,0)\cap J(0,0,1)=(xyz)$. Since every
global decomposition restricts to the box, the minimal number of components is
exactly $3$. (Conceptually: a principal squarefree monomial $(m)$ has the
irredundant irreducible decomposition $(m)=\bigcap_{x_i\mid m}(x_i)$ with
$|\operatorname{supp}(m)|$ components; here $|\operatorname{supp}(xyz)|=3$.)
\end{proof}

\begin{theorem}\label{thm:beta0}
$\beta_0\big((xyz)\big)=1$.
\end{theorem}

\begin{proof}
$(xyz)$ is generated by the single monomial $xyz$, and it is not the zero
ideal since $xyz\in(xyz)$; the minimal monomial generating set is
$\{xyz\}$ (also confirmed by brute force, \texttt{reproduce.py} check~3).
\end{proof}

\begin{theorem}[Main]\label{thm:main}
$\iota\big((xyz)\big)=3>1=\beta_0\big((xyz)\big)$. Consequently TLMC
conjecture 00000000548 is \textbf{false}.
\end{theorem}

\begin{proof}
Immediate from Theorems~\ref{thm:iota} and \ref{thm:beta0}: the conjectured
inequality $\iota(I)\le\beta_0(I)$ fails for $I=(xyz)$ in $k[x,y,z]$, i.e.\ at
$n=3$.
\end{proof}

\section{Remarks on the boundary}

\begin{remark}
The counterexample lies inside the range the conjecture declares verified
($n=3\le12$), and $(xyz)$ is itself a complete intersection (one generator),
so the rider ``tightness attained by complete intersections'' fails as well:
a complete intersection can have $\iota=3\neq1=\beta_0$.
\end{remark}

\begin{remark}
For the squarefree principal monomial $(x_1x_2\cdots x_t)$ one has
$\iota=t$ and $\beta_0=1$. Hence the inequality fails for every $t\ge2$ and
$\iota/\beta_0$ is unbounded already on principal ideals: no repair of the
form $\iota\le\beta_0$ is possible on any class containing these examples.
\end{remark}

\begin{remark}
The correct combinatorial picture is that $\iota(I)$ of a monomial ideal
equals its number of maximal standard pairs, an invariant not controlled by
the number of generators. Any corrected statement must bound $\iota(I)$ by a
different quantity (e.g.\ the number of facets of an associated simplicial
complex).
\end{remark}

\section{Verification artifacts}

\begin{itemize}
\item \texttt{reproduce.py} recomputes every number from scratch by finite
enumeration in the box $[0,4]^3$: set equality, irredundancy witnesses,
$\beta_0=1$ by brute force over generator sets, irreducibility of the three
components by exhaustive search over all $252$ box super-ideals of each, and
minimality of the component count over all $216$ irreducible-form ideals.
Output: $\iota=3$, $\beta_0=1$.
\item \texttt{lean4/} contains an axiom-free Lean~4 formalization (toolchain
\texttt{leanprover/lean4:v4.33.1}, no Mathlib). \texttt{Main.lean} proves the
decomposition $(xyz)=(x)\cap(y)\cap(z)$ (pointwise, by case analysis),
irredundancy via the witnesses $yz,xz,xy$, the irreducibility lemma
(constructively, using $u+v$ instead of $uv$ to avoid case analysis on natural
numbers), $\beta_0=1$, and the numeric conclusion $3>1$.
\texttt{Check.lean} audits all theorems with \texttt{\#print axioms}: every
one reports \emph{does not depend on any axioms}.
\item \texttt{build/main.pdf} is this document;
\texttt{tectonic main.tex -{}-outdir build} rebuilds it.
\end{itemize}
