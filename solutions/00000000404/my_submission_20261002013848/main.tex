\begin{abstract}
Conjecture 00000000404 asserts that in the Schur expansion of the plethysm
$\s{(2)}[\s{(n)}]$, the coefficients indexed by even partitions $\nu$ are all
even.  We exhibit the counterexample $n=2$:
\[
\s{(2)}[\s{(2)}] \;=\; \s{(4)} + \s{(2,2)},
\]
whose two nonzero Schur coefficients are both $1$, hence odd.  Under every
natural reading of ``even-indexed $\nu$'' ($|\nu|$ even, all parts even,
first part even, or even length) this violates the conjecture.  The identity
is derived in the power-sum basis, cross-checked by two independent Weyl
dimension counts, and fully machine-verified in core Lean with zero axioms.
\end{abstract}

\section{The conjecture}

\begin{quote}
\textbf{00000000404.} \emph{Definition: Schur plethysm $\s\lambda \circ \s\mu$.
Conjecture: among the Schur expansion coefficients of $\s{(2)}[\s{(n)}]$, the
coefficients of even-indexed $\nu$ are all even.}
\end{quote}

The phrase ``even-indexed $\nu$'' is ambiguous; we refute it under all four
natural readings simultaneously (Section~\ref{sec:readings}).

\section{The attack at $n=2$}

Recall the $\lambda$-ring identity for the second symmetric power: for any
symmetric function $f$,
\[
\s{2}[f] \;=\; \tfrac{1}{2}\bigl(f\cdot f + \psi^2(f)\bigr),
\qquad \psi^2(p_k) = p_{2k},
\]
since $\s{2} = (p_1^2 + p_2)/2$ and plethysm is a composition of
$\Lambda$-ring endomorphisms.  With $f = \s{2} = (p_{11} + p_2)/2$ we get
\[
\s{2}\cdot\s{2} = \frac{(p_{11}+p_2)^2}{4}
 = \frac{p_{1111} + 2p_{211} + p_{22}}{4},
 \qquad
 \psi^2(\s{2}) = \frac{p_{22}+p_4}{2},
\]
hence
\begin{equation}\label{eq:pbasis}
\s{(2)}[\s{(2)}] \;=\; \frac{p_{1111} + 2\,p_{211} + 3\,p_{22} + 2\,p_4}{8}.
\end{equation}
(The subscripts record part multiplicities, e.g.\ $p_{211} = p_2p_1^2$.)

Schur functions expand as $\s\nu = \sum_{\rho} \chi^{\nu}(\rho)\, p_\rho /
z_\rho$, where $\chi^\nu$ is the irreducible character of $S_{|\nu|}$ at
cycle type $\rho$ and $z_\rho = \prod_i i^{m_i} m_i!$.  By Hall orthogonality
$\langle p_\rho, \s\nu\rangle = \chi^\nu(\rho)$, the Schur coefficients of
\eqref{eq:pbasis} are $c_\nu = \sum_\rho a_\rho \chi^\nu(\rho)$ with
$a = (1,2,0,3,2)/8$ over $(p_{1111},p_{211},p_{31},p_{22},p_4)$.  The $S_4$
character table (columns $= 1^4,\, 21^2,\, 31,\, 22,\, 4$; verified by the
Murnaghan--Nakayama rule in \texttt{reproduce.py}) is:

\begin{center}
\begin{tabular}{lccccc}
\toprule
$\nu \backslash \rho$ & $1111$ & $211$ & $31$ & $22$ & $4$\\
\midrule
$(4)$      & $1$ & $1$ & $1$  & $1$  & $1$\\
$(31)$     & $3$ & $1$ & $0$  & $-1$ & $-1$\\
$(22)$     & $2$ & $0$ & $-1$ & $2$  & $0$\\
$(211)$    & $3$ & $-1$& $0$  & $-1$ & $1$\\
$(1111)$   & $1$ & $-1$& $1$  & $1$  & $-1$\\
\bottomrule
\end{tabular}
\end{center}

Dotting $a$ with each row:
\[
\begin{aligned}
8\,c_{(4)} &= 1+2+0+3+2 = 8, &
8\,c_{(31)} &= 3+2+0+0+0 \;\;&&=\; 0,\\
8\,c_{(22)} &= 2+0+0+6+0 = 8, &
8\,c_{(211)} &= 3-2+0-3+2 \;&&=\; 0,\\
&& 8\,c_{(1111)} &= 1-2+0+3-2 \;&&=\; 0.
\end{aligned}
\]
Therefore
\begin{equation}\label{eq:attack}
\boxed{\;\s{(2)}[\s{(2)}] \;=\; \s{(4)} + \s{(2,2)},\qquad
c_{(4)} = 1,\; c_{(2,2)} = 1\;}
\end{equation}
and both nonzero coefficients are \textbf{odd}.

\subsection{Dimension cross-check}\label{sec:dims}

For $d=4$: $\dim \Sym^2(\Sym^2 \mathbb{C}^4) = \binom{10+1}{2} = 55$, while
$\dim S_{(4)}\mathbb{C}^4 = \binom{4+4-1}{4} = 35$ and
$\dim S_{(2,2)}\mathbb{C}^4 = 20$ (hook--content / Weyl formula).  The
multiplicity-free decomposition \eqref{eq:attack} matches the count
$55 = 35 + 20$.  For $d=2$ likewise
$\dim \Sym^2(\Sym^2\mathbb{C}^2) = \binom{4}{2} = 6 = 5 + 1$
($\dim \s{(4)} = 5$, $\dim \s{(2,2)} = 1$), again consistent.

\section{Every reading of ``even-indexed'' is refuted}\label{sec:readings}

\begin{center}
\begin{tabular}{lll}
\toprule
Reading of ``even-indexed $\nu$'' & Witness $\nu$ at $n=2$ & Coefficient\\
\midrule
$|\nu|$ even          & $(4)$ and $(2,2)$ & $1$ (odd)\\
all parts even        & $(4)$ and $(2,2)$ & $1$ (odd)\\
first part even       & $(4)$ and $(2,2)$ & $1$ (odd)\\
length $\ell(\nu)$ even & $(2,2)$         & $1$ (odd)\\
\bottomrule
\end{tabular}
\end{center}

In every case the coefficient is $1 \not\equiv 0 \pmod 2$.  The conjecture is
false.

\section{Boundary and scope}

\begin{itemize}
\item \textbf{$n=1$:} $\s{(2)}[\s{(1)}] = \s{(2)}$ with coefficient $1$
(odd); $\nu=(2)$ has even size, all parts even, first part even.  The failure
is not an $n=2$ accident.
\item \textbf{General $n$ (computed for $n \le 6$):} the expansion is
multiplicity-free,
$\s{(2)}[\s{(n)}] = \sum_{j=0}^{\lfloor n/2\rfloor} \s{(2n-2j,\; 2j)}$,
every coefficient being exactly $1$; each summand has even size, all parts
even and first part even, and for $n\ge 2$ also even length.  So no choice of
$n$ rescues the conjecture.
\end{itemize}

\section{Machine verification}

The directory \texttt{lean4/} contains a core-Lean (no Mathlib) formalization
(\texttt{lakefile.toml}, \texttt{lean-toolchain} \texttt{leanprover/lean4:v4.33.1}):

\begin{itemize}
\item \texttt{attack\_expansion}: identity \eqref{eq:pbasis}, proved by
\texttt{rfl} on the explicit $\lambda$-ring computation;
\item \texttt{character\_table\_orthogonal}: the $S_4$ table above is
column-orthogonal (weights $1,6,8,3,6$), all 25 pairs checked by
\texttt{decide};
\item \texttt{conjecture\_404\_false} / \texttt{parity\_odd}: the five
coefficients of \eqref{eq:attack} and their parity, by \texttt{decide};
\item \texttt{dim\_check}: $55 = 35 + 20$; \quad
\texttt{boundary\_n1}: the $n=1$ case.
\end{itemize}

The axiom audit (\texttt{lake env lean Check.lean}) reports that all six
theorems \emph{do not depend on any axioms}: no \texttt{sorry}, no
\texttt{propext}, no \texttt{Quot.sound}, no \texttt{Classical.choice}, no
\texttt{Lean.ofReduceBool}.
