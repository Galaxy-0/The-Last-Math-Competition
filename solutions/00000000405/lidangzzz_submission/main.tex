\section{Conjecture \#405: divisibility of the Kostka spin-pairing product}

\begin{conjecture}[TLMC-00000000405]\label{conj:405}
For every partition $\lambda \vdash n$, the spin-pairing product $K_{\lambda,(1^n)} \cdot K_{\lambda',(1^n)}$ divides $n!$.
\end{conjecture}

\begin{theorem}
Conjecture~\ref{conj:405} is \textbf{false}.
\end{theorem}

\begin{proof}
Take $n = 3$ and $\lambda = (2,1)$, which is self-conjugate ($\lambda' = (2,1)$). By the standard combinatorial definition, $K_{\lambda,(1^n)}$ counts semistandard fillings of shape $\lambda$ with content $(1,\dots,1)$, i.e.\ fillings of the diagram
\[
\begin{array}{cc} a & b \\ c & \end{array}
\]
by a permutation of $\{1,2,3\}$ with weakly increasing rows ($a \le b$) and strictly increasing columns ($a < c$). Exactly two fillings qualify: $(a,b,c) = (1,2,3)$ and $(1,3,2)$. Hence both factors equal $2$, the product is $2 \cdot 2 = 4$, and $4 \nmid 3! = 6$.
\end{proof}
