@CompleteGraph.hasQuadraticMarkovBasis : ∀ {V : Type u_1} [inst : DecidableEq V] {k : Type u_2} [inst_1 : CommRing k]
  [inst_2 : Nontrivial k] [inst_3 : Fintype V], CompleteGraph.HasQuadraticMarkovBasis V k
def CompleteGraph.HasQuadraticMarkovBasis.{u_1, u_2} : (V : Type u_1) →
  [inst : DecidableEq V] → (k : Type u_2) → [inst : CommRing k] → Prop :=
fun V [DecidableEq V] k [CommRing k] =>
  ∃ B,
    Ideal.span ↑B = CompleteGraph.toricIdeal V k ∧
      ∀ p ∈ B,
        p.totalDegree = 2 ∧
          ∃ a b,
            a ≠ b ∧
              (a.sum fun x n => n) = 2 ∧
                (b.sum fun x n => n) = 2 ∧ p = (MvPolynomial.monomial a) 1 - (MvPolynomial.monomial b) 1
def CompleteGraph.graphRingMap.{u_1, u_2} : (V : Type u_1) →
  [inst : DecidableEq V] →
    (k : Type u_2) → [inst : CommRing k] → MvPolynomial (CompleteGraph.Edge V) k →+* MvPolynomial V k :=
fun V [DecidableEq V] k [CommRing k] => IdealBridge.toricMap (CompleteGraph.incidence V)
def CompleteGraph.toricIdeal.{u_1, u_2} : (V : Type u_1) →
  [inst : DecidableEq V] → (k : Type u_2) → [inst : CommRing k] → Ideal (MvPolynomial (CompleteGraph.Edge V) k) :=
fun V [DecidableEq V] k [CommRing k] => RingHom.ker (CompleteGraph.graphRingMap V k)
