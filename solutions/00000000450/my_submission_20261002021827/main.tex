\begin{claim*}[TLMC 00000000450]
For the $q,t$-Catalan number $C_n(q,t)$ and every relevant root of unity $\zeta$
(primitive $n$-th root of unity),
\[
  C_n(\zeta,\zeta^{-1}) = (n+1)^{(n-1)/2}.
\]
\end{claim*}

\begin{quote}\textbf{Verdict: FALSE.}
Under the standard reading of $C_n(q,t)$ (the $q,t$-Catalan number; cf.\ the
neighbouring corpus entry 00000000449, which describes ``the $q,t$-Catalan'' as
``Haiman's bisymmetric generalization of the Catalan numbers''), the conjecture
holds at $n=1$ and fails at every $n=2,\dots,6$: the true value is exactly $-2$
for even $n$ and $+2$ for odd $n$, while $(n+1)^{(n-1)/2}=\sqrt3,4,5^{3/2},36,7^{5/2}$
in turn. The refutation is verified by independent exact recomputation
(Python, pure integer arithmetic) and by a zero-axiom Lean~4 formalization.
\end{quote}

\section{Definition and setup}

We use the standard \emph{dinv--area} form of the $q,t$-Catalan number
(Garsia--Haiman; Haglund). A Dyck path of semilength $n$ is encoded by its
\emph{area sequence} $a=(a_1,\dots,a_n)$, characterized by
\[
  a_1 = 0,\qquad 0 \le a_{i+1} \le a_i + 1 \quad (1\le i<n),
\]
and there are Catalan-many such sequences. The two classical statistics are
\[
  \operatorname{area}(D)=\sum_{i=1}^n a_i,
  \qquad
  \operatorname{dinv}(D)=\#\{i<j : a_i=a_j\} + \#\{i<j : a_i=a_j+1\}
\]
(primary and secondary inversion pairs). Then
\begin{equation}
  C_n(q,t) \;=\; \sum_{D\in\mathcal{D}_n} q^{\operatorname{area}(D)}\,t^{\operatorname{dinv}(D)}.
  \label{eq:qtc}
\end{equation}
Sanity checks: $C_2(q,t)=q+t$, so $C_2(1,1)=2$, and $C_3(1,1)=5$: the Catalan numbers.

At $q=\zeta$, $t=\zeta^{-1}$ (with $\zeta$ a primitive $n$-th root of unity, so
$\zeta^n=1$) every monomial collapses:
\begin{equation}
  C_n(\zeta,\zeta^{-1}) \;=\; \sum_{D\in\mathcal{D}_n}
  \zeta^{\,\operatorname{area}(D)-\operatorname{dinv}(D)},
  \label{eq:root}
\end{equation}
an \emph{exact cyclotomic integer}, computable without any floating point.

\section{The attacks}

\paragraph{$n=2$ (pipeline anchor reproduced).}
$\mathcal{D}_2$ consists of the two area sequences $(0,0)$ and $(0,1)$ with
$(\operatorname{area},\operatorname{dinv})$ equal to $(0,1)$ and $(1,0)$, so
$C_2(q,t)=q+t$ and
\[
  C_2(-1,-1) = -1 + -1 = -2 \;\ne\; 3^{1/2}=\sqrt3 .
\]
($\sqrt3$ is irrational, hence in particular not the integer $-2$; equivalently
$(-2)^2=4\ne 3=(\sqrt3)^2$.)

\paragraph{$n=3$ (pipeline anchor reproduced; integer vs.\ integer).}
The five Dyck paths have $(\operatorname{area},\operatorname{dinv})$
\[
  (0,3),\ (1,1),\ (1,2),\ (2,1),\ (3,0),
\]
so the exponents $\operatorname{area}-\operatorname{dinv}$ are $-3,0,-1,1,3$ and,
for a primitive cube root $\zeta$ (using $\zeta+\zeta^{-1}=-1$),
\[
  C_3(\zeta,\zeta^{-1}) = \zeta^{-3}+\zeta^{0}+\zeta^{-1}+\zeta^{1}+\zeta^{3}
  = 3 + (\zeta+\zeta^{-1}) = 2 \;\ne\; 4=(3+1)^{(3-1)/2}.
\]

\paragraph{$n=5$ (integer vs.\ integer).}
Exact evaluation over all $42$ Dyck paths gives, in $\mathbb{Z}[\zeta]/(\Phi_5)$,
\[
  C_5(\zeta,\zeta^{-1}) = 2 \;\ne\; 36 = 6^{(5-1)/2}.
\]

\paragraph{$n=4,6$.} Exact evaluation gives $C_4(i,i^{-1})=-2$ (in
$\mathbb{Z}[i]$) and $C_6(\zeta_6,\zeta_6^{-1})=-2$ (in $\mathbb{Z}[\zeta_6]$),
while the claims $5^{3/2}$ and $7^{5/2}$ are irrational.

\begin{theorem}[Exact values, $1\le n\le 6$]
\begin{center}
\begin{tabular}{ccc}
\toprule
$n$ & $C_n(\zeta_n,\zeta_n^{-1})$ (exact) & claimed $(n+1)^{(n-1)/2}$ \\
\midrule
1 & $1$ & $1$ (holds)\\
2 & $-2$ & $\sqrt3$ \\
3 & $2$ & $4$ \\
4 & $-2$ & $5^{3/2}$ \\
5 & $2$ & $36$ \\
6 & $-2$ & $7^{5/2}$ \\
\bottomrule
\end{tabular}
\end{center}
\end{theorem}

In every case $n\ge2$ the two sides disagree: for odd $n$ both sides are
ordinary integers and $2\ne(n+1)^{(n-1)/2}$; for even $n$ the left side is the
algebraic integer $\pm2$ while the right side is irrational
($n+1$ is not a perfect square for $2\le n\le 6$). Hence the universal claim
of Conjecture 00000000450 is \textbf{false}.

\section{Verification}

\begin{itemize}
  \item \textbf{Independent recomputation.} \texttt{reproduce.py} (stdlib-only,
  pure integer arithmetic) enumerates all Dyck area sequences, computes
  $\operatorname{area}-\operatorname{dinv}$, and reduces
  $\sum_D \zeta^{\operatorname{area}-\operatorname{dinv}}$ in
  $\mathbb{Z}[\zeta]/(\Phi_n)$. It reproduces both pipeline anchors
  ($C_2=q+t$, $C_2(-1,-1)=-2$; $C_3(\zeta,\zeta^{-1})=2$) and the full table above.
  Run: \texttt{python3 reproduce.py}.
  \item \textbf{Lean 4 formalization (zero axioms).} \texttt{lean4/Main.lean}
  defines the area-sequence enumerator and both statistics in core Lean~4
  (no Mathlib), and proves by kernel computation (\texttt{rfl}/\texttt{decide}):
  \texttt{C2 = -2}, \texttt{C3 = 2}, \texttt{C3 $\ne$ 4}, \texttt{C5 = 2},
  \texttt{C5 $\ne$ 36}, among others. \texttt{lean4/Check.lean} runs
  \texttt{\#print axioms} on every theorem; all report
  \emph{``does not depend on any axioms''}.
\end{itemize}

\section{Boundary}

At $n=1$ the unique Dyck path has $(\operatorname{area},\operatorname{dinv})=(0,0)$
and $C_1(1,1)=1=(1+1)^{0}$, so the conjecture is consistent at $n=1$; it fails
already at $n=2$ and at every larger $n$ tested ($2\le n\le 6$).
