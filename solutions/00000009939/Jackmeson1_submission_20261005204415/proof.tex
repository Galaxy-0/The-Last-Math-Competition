\section{The conjecture and the clause refuted}

\textbf{Statement (English, verbatim).} Definition: Persistence entropy: the Shannon functional of
persistence bar length distributions. Conjecture: The persistence entropy satisfies concentration
inequalities and is mutually bounded with the Shannon entropy of the longest bar: the entropy minus
longest-bar contribution is bounded by half the logarithm of the bar count; the constant of the bound
is tight.

\textbf{Chinese text (translated).} The Chinese version says that the persistence entropy is
mutually bounded with the Shannon entropy of the longest bar, and makes this precise as: ``the
entropy minus the longest bar's contribution is bounded by half of the logarithm of the number of
bars; the constant of the bound is tight.''

\textbf{Reading.} The conjecture is a conjunction. Its explicit quantitative conjunct is
\[
  \text{(B)}\qquad E(B)-c_B(I)\ \le\ \tfrac12\log n,
\]
for every nonempty barcode $B$ with $n$ bars and every longest bar $I$ of $B$,
where $E$ is the persistence entropy and $c_B(I)$ is the contribution of $I$ to $E$, i.e.\ its own
summand $-p_I\log p_I$. We refute (B), which refutes the conjunction. We do not address the vague
clause ``satisfies concentration inequalities''. The tightness clause presupposes (B); we also show
that no constant $c<1$ can replace $\tfrac12$ (Theorem~\ref{thm:const}).

\textbf{Conventions.} Natural logarithm. A change of base multiplies both sides of (B) by the same
positive factor, so the result does not depend on the base. $\nml(x)=-x\log x$ (Mathlib
\texttt{Real.negMulLog}). Bars are counted with multiplicity; in all our witnesses the bars are
pairwise distinct intervals, so counting distinct bars gives the same $n$. All witnesses have a
\emph{unique} longest bar, so the choice of ``the longest bar'' is not an issue.

\section{Definitions (as formalized)}

\begin{definition}
A \emph{bar} is a finite interval $[b,d)$ with real $b<d$; its length is $\ell=d-b>0$. A
\emph{barcode} is a finite multiset of bars. For a barcode $B$ with lengths $\ell_1,\dots,\ell_n$,
put $L=\sum_i\ell_i$ and $p_i=\ell_i/L$. The \emph{persistence entropy} is
\[
  E(B)=-\sum_{i=1}^n p_i\log p_i=\sum_{i=1}^n \nml(\ell_i/L),
\]
and the \emph{contribution} of a bar $I$ of length $\ell_I$ is $c_B(I)=\nml(\ell_I/L)$. A bar $I\in B$
is a \emph{longest bar} if $\ell_J\le\ell_I$ for all $J\in B$.
\end{definition}

This is the Shannon entropy of the distribution $(p_i)$ given by the normalized bar lengths, which is
the standard notion. Atienza, Gonzalez-Diaz and Rucco (arXiv:1701.07857) write: ``Persistent entropy
is defined as the Shannon entropy of the persistence barcode of a given filtration.'' Atienza,
Gonzalez-Diaz and Soriano-Trigueros (arXiv:1803.08304) note: ``Observe that, to compute persistent
entropy, we only have to consider the length of each interval'', and ``in the definition of
persistent entropy, we assume that there are no infinite length intervals in the persistence
barcode.'' This is why bars are finite here.

\textbf{Realization by a filtration.} Every finite multiset of finite bars $[b_j,d_j)$,
$j=1,\dots,k$, is the degree-1 persistence barcode of a filtered simplicial complex. Take $k$ hollow
triangles $T_j$ that share one vertex $v$. Let $v$ enter at time $\min_j b_j$, let the other two
vertices and the three edges of $T_j$ enter at time $b_j$, and let the 2-simplex filling $T_j$ enter
at time $d_j$. At time $t$, $H_1$ is freely generated by the classes $[\partial T_j]$ with
$b_j\le t<d_j$, and each inclusion map sends $[\partial T_j]$ to itself (or to $0$ once $T_j$ is
filled). Hence $H_1$ is the direct sum of the interval modules on $[b_j,d_j)$. This construction is
prose only; Mathlib has no persistent homology, so the Lean input is the barcode itself.

\section{The witnesses}

For $a>0$ and $m\in\mathbb{N}$ let $S(a,m)$ be the barcode consisting of $[0,a)$ and the $m$ unit bars
$[i,i+1)$, $0\le i<m$ (Lean \texttt{spike a ha m}). It has $n=m+1$ bars and total length $L=a+m$.

\begin{lemma}[\texttt{entropy\_spike}, \texttt{longest\_eq}, \texttt{gap\_spike}, \texttt{spike\_nodup}]
$E(S(a,m))=\nml(a/L)+m\,\nml(1/L)$. If $a\ne1$ the $n$ bars are pairwise distinct. If $a>1$ then
$[0,a)$ is the unique longest bar and
\[
  E(S(a,m))-c(\,[0,a)\,)=m\,\nml(1/L)=\frac{m}{a+m}\log(a+m).
\]
\end{lemma}

\begin{proof}
The contribution of $[0,a)$ is $\nml(a/L)$ and each unit bar contributes $\nml(1/L)$, so the formula
for $E$ follows. A unit bar has length $1<a$, so it is not longest. Finally
$\nml(1/L)=-(1/L)\log(1/L)=(\log L)/L$.
\end{proof}

\begin{theorem}[\texttt{bound\_fails\_every\_count}]\label{thm:main}
For every $n\ge2$, the barcode $S(2,n-1)$ has exactly $n$ pairwise distinct bars and a unique longest
bar $I$, and $E-c(I)=\frac{n-1}{n+1}\log(n+1)>\frac12\log n$.
\end{theorem}

\begin{proof}
Write $m=n-1\ge1$ (Lean \texttt{key\_ineq}). If $m=1$, the claim is $\frac13\log3>\frac12\log2$,
i.e.\ $\log 9>\log 8$. If $m\ge2$, then $\frac{m}{m+2}\ge\frac12$ and $\log(m+2)>\log(m+1)\ge0$, so
$\frac{m}{m+2}\log(m+2)\ge\frac12\log(m+2)>\frac12\log(m+1)$.
\end{proof}

\textbf{The smallest example with three bars} (\texttt{witness\_211}). The bars $[0,2),[0,1),[1,2)$
have lengths $(2,1,1)$, $L=4$ and $p=(\frac12,\frac14,\frac14)$. Then $E=\frac32\log2$, the longest
bar contributes $\frac12\log2$, and $E-c=\log2>\frac12\log3$, because $4>3$.

\begin{theorem}[\texttt{conjecture9939\_false}]
Statement (B) is false.
\end{theorem}

\begin{proof}
Apply Theorem~\ref{thm:main} with $n=3$.
\end{proof}

\begin{theorem}[\texttt{no\_constant\_below\_one}]\label{thm:const}
For every real $c<1$ there is a barcode, with a longest bar, such that $E-c(I)>c\log n$ for
every longest bar $I$.
\end{theorem}
(The Lean statement concludes exactly this. The witness used is the barcode \texttt{spike 2}
plus unit bars, whose longest bar is unique by the separately proved lemma
\texttt{longest\_eq}; uniqueness is therefore established in the development, though not
stated in this theorem's conclusion.)

\begin{proof}
Choose $m\in\mathbb{N}$ with $m>2/(1-c)$. Then $m(1-c)>2>2c$, so $\frac{m}{m+2}>c$. Since
$\log(m+2)>\log(m+1)>0$, we get $\frac{m}{m+2}\log(m+2)>\frac{m}{m+2}\log(m+1)>c\log(m+1)$.
\end{proof}

\section{Alternative readings of ``the Shannon entropy of the longest bar''}

The Chinese text fixes the reading ``contribution'' $=c(I)$. Two other readings of the English
phrase are also refuted, by the barcode $S(16,48)$ (one bar $[0,16)$ and $48$ unit bars, $n=49$,
$L=64$, $p_{\max}=\frac14$).
\begin{itemize}
\item \emph{Binary entropy} $h(p_{\max})=\nml(p_{\max})+\nml(1-p_{\max})$ (Mathlib
  \texttt{Real.binEntropy}). Lemma \texttt{binGap\_spike} gives
  $E-h(p_{\max})=\frac{m}{L}\log m=\frac34\log48$, and $\frac34\log48>\frac12\log49$ because
  $48^3>49^2$ (\texttt{binary\_reading\_false}).
\item \emph{Self-information / min-entropy} $-\log p_{\max}$. Lemma \texttt{minGap\_spike} gives
  $E+\log p_{\max}=\frac{m}{L}\log a=\frac34\log16=\log 8$, and $\log8>\log7=\frac12\log49$
  (\texttt{minEntropy\_reading\_false}).
\end{itemize}
Since all gaps above are positive, the bound also fails if ``bounded by'' is read as
$|E-c(I)|\le\frac12\log n$.

\begin{remark}[not formalized]
If ``bar length distribution'' meant instead the empirical distribution of the length \emph{values}
(a histogram), the bars $[0,1),[0,2),[0,3)$ would give a uniform histogram on three values, with
entropy $\log3$ and longest-bar contribution $\frac13\log3$. The difference $\frac23\log3$ again
exceeds $\frac12\log3$. This reading is non-standard and is not part of the Lean development.
\end{remark}
