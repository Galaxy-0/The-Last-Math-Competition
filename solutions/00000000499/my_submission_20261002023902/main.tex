\title{Disproof of Conjecture 00000000499}
\author{}
\date{October 2, 2026}
\maketitle

\begin{abstract}
\noindent\textbf{Verdict: FALSE.} The conjecture asserts that the vicious-walker
hitting probability $p_n$ equals, at the corresponding endpoints, the joint density
of the ordered eigenvalues of an $n\times n$ GUE matrix times the explicit
Vandermonde factor $\prod_{i<j}\bigl(y_j-y_i-(x_j-x_i)\bigr)$, and that this
identity holds for all times $T>0$. The right-hand side contains no $T$ and is
therefore a single constant, while at the fixed endpoints $(x_1,x_2)=(0,4)$,
$(y_1,y_2)=(1,3)$ for $n=2$ an exact brute-force enumeration gives the distinct
values $p(1)=1/4$, $p(2)=0$ and $p(3)=1/8$. A constant cannot equal three different
numbers, so the identity cannot hold for all $T>0$.
\end{abstract}

\section{The conjecture}

Let $p_n$ denote the probability that $n$ nonintersecting simple random walks go
from fixed starting points $x=(x_1,\dots,x_n)$ to fixed endpoints
$y=(y_1,\dots,y_n)$ without ever intersecting. The conjecture states that
\[
p_n \;=\; \rho^{\mathrm{GUE}}_{n,\mathrm{ord}}(y)\;\cdot\!\!\prod_{i<j}\bigl(y_j-y_i-(x_j-x_i)\bigr),
\]
where $\rho^{\mathrm{GUE}}_{n,\mathrm{ord}}$ is the joint density of the ordered
eigenvalues of an $n\times n$ GUE matrix, and that this holds \emph{for all times
$T>0$}.

\section{Attack: same endpoints, different times}

Take $n=2$ with starts $x=(0,4)$ and endpoints $y=(1,3)$; each walk moves
$\pm1$ per step, and all $4^T$ joint step sequences are equally likely. We
enumerate every joint step sequence exactly and count those for which both walks
end at $(1,3)$ and the strict order $w_1(t)<w_2(t)$ holds at every time
$t=0,\dots,T$.

\begin{center}
\begin{tabular}{c c c}
\hline
$T$ & good / total & $p(T)$\\
\hline
1 & $1/4$      & $1/4$\\
2 & $0/16$     & $0$\\
3 & $8/64$     & $1/8$\\
4 & $0/256$    & $0$\\
5 & $75/1024$  & $75/1024$\\
\hline
\end{tabular}
\end{center}

\begin{itemize}
\item $T=1$: the only surviving joint path is $(+1,-1)$, taking $(0,4)$ to $(1,3)$
  in one step; $p(1)=\tfrac14$.
\item $T=2$: parity. A displacement of $+1$ in two $\pm1$ steps is impossible, so
  no joint path reaches the endpoints; $p(2)=0$.
\item $T=3$: exactly $8$ of the $64$ joint paths survive; for instance
  $w_1\colon 0\to1\to0\to1$ with $w_2\colon 4\to3\to4\to3$.
  Hence $p(3)=\tfrac{8}{64}=\tfrac18$.
\end{itemize}

\section{Why the conjecture fails}

The proposed right-hand side evaluated at these endpoints is
\[
\rho^{\mathrm{GUE}}_{2,\mathrm{ord}}(1,3)\cdot
\Bigl((3-1)-(4-0)\Bigr)
\;=\;-2\,\rho^{\mathrm{GUE}}_{2,\mathrm{ord}}(1,3),
\]
which \emph{does not depend on $T$}. It is a single fixed constant $c$; moreover
$c\neq 0$, since the ordered GUE density at two distinct points is strictly
positive and the Vandermonde-type factor equals $\pm2\neq0$ (under the literal
reading $c$ is even negative, while a probability satisfies $p_n\ge 0$).

The left-hand side, however, takes the distinct values $\tfrac14$, $0$, $\tfrac18$
at the same endpoints as $T$ varies. A single constant cannot equal three different
numbers; in particular $p(1)=\tfrac14\neq\tfrac18=p(3)$, while the right-hand side
is the same number at $T=1$ and $T=3$. Therefore the identity fails for some
$T>0$, and the conjecture is \textbf{false}.

\section{Verification}

\begin{itemize}
\item \texttt{reproduce.py}: standalone exact recomputation (standard library
  only), recomputing the table above with rational arithmetic.
\item \texttt{lean4/}: a core-Lean (no Mathlib) development that explicitly
  enumerates the $4$, $16$ and $64$ joint step sequences for $T=1,2,3$ and proves,
  with zero axioms and no \texttt{sorry}, the counts $C_1=1$, $C_2=0$, $C_3=8$ and
  the cross-multiplied inequality $C_1\cdot 64\neq C_3\cdot 4$ (i.e.
  $\tfrac14\neq\tfrac18$). An axiom audit (\texttt{Check.lean}) confirms each
  theorem ``does not depend on any axioms''.
\end{itemize}
