\begin{abstract}
We disprove a conjecture of the Last Math Competition asserting that for the binomial matroid $\mathrm{bin}(n,i)$ — the binary matroid on the $\binom{n}{i}$ incidence vectors of weight $i$ in $\mathbb{F}_2^n$, i.e.\ the $i$-subsets under symmetric difference (for $i = 2$ this is the graphic matroid $M(K_n)$) — the value $\chi_{n,i}(-m)$ of the characteristic polynomial is divisible by $i!\,(n-i)!$ for all $2 \le i \le n-2$ and $m \ge 1$. The smallest case already fails: $\chi_{4,2}(t) = t^3 - 6t^2 + 11t - 6$ and $\chi_{4,2}(-4) = -210$, while $2!\cdot 2! = 4 \nmid 210$. The computation is formalized in Lean~4 via Whitney's subset expansion with the linear rank computed by brute force over all $2^6$ coefficient vectors, entirely kernel-verified by \texttt{decide}.
\end{abstract}

\section{The conjecture}

\begin{conjecture}[TLMC-00000000578]\label{conj:578}
Let $\mathrm{bin}(n,i)$ be the binomial matroid (the $i$-subsets of $\{1,\dots,n\}$ under symmetric difference, i.e.\ the binary matroid of the weight-$i$ vectors of $\mathbb{F}_2^n$) and $\chi_{n,i}$ its characteristic polynomial. For all $2 \le i \le n-2$ and integers $m \ge 1$, $\chi_{n,i}(-m)$ is divisible by $i!\,(n-i)!$, with quotient a nonnegative combination of Eulerian numbers.
\end{conjecture}

\section{The disproof}

\begin{definition}[Characteristic polynomial]
For a matroid $M$ on ground set $E$ with rank function $r$, Whitney's subset expansion is
\[
\chi_M(t) \;=\; \sum_{A \subseteq E} (-1)^{|A|}\, t^{\,r(E) - r(A)} .
\]
For a binary matroid given by vectors, $r(A)$ is the $\mathbb{F}_2$-dimension of the span of $A$, equivalently the maximum size of a linearly independent subfamily.
\end{definition}

\begin{theorem}\label{thm:578}
Conjecture~\ref{conj:578} is \textbf{false}. At $n = 4$, $i = 2$ (the six weight-2 vectors of $\mathbb{F}_2^4$, i.e.\ the edges of $K_4$, spanning the rank-3 even-weight subspace):
\[
\chi_{4,2}(t) \;=\; t^3 - 6t^2 + 11t - 6 \qquad\left(= \frac{P_{K_4}(t)}{t}\right),
\]
so $\chi_{4,2}(-4) = -64 - 96 - 44 - 6 = -210$, and $i!\,(n-i)! = 2!\cdot 2! = 4$ does not divide $210$.
\end{theorem}

\begin{proof}
Kernel computation (Lean \texttt{decide}). The rank of each of the $64$ subsets of the ground set is computed as the maximum size of a linearly independent subfamily, independence being checked by brute force over all $2^6$ coefficient vectors of $\mathbb{F}_2^6$; the sanity theorem \texttt{linIndepB\_sanity} exhibits the tester behaving correctly on $\varnothing$, singletons, an independent pair, and the dependent triple $\{v_0, v_1, v_3\}$ (where $v_0 + v_1 = v_3$). The subset expansion then yields the four coefficients of $\chi$ and its value at $-4$.
\end{proof}

\begin{remark}
The companion clause (the quotient being a nonnegative combination of Eulerian numbers) fails with the divisibility. For comparison, the cases $(n,i) = (5,2)$ and $(7,2)$ pass the divisibility for all $m \le 39$ in our computation, so the failure at $(4,2)$ is a genuine small-case refutation, not a misreading that fails everywhere.
\end{remark}
