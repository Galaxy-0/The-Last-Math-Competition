@[reducible] def Conjecture236.SignMatrix : ℕ → Type :=
fun n => Matrix (Fin n) (Fin n) Bool
def Conjecture236.integerMatrix : {n : ℕ} → Conjecture236.SignMatrix n → Matrix (Fin n) (Fin n) ℤ :=
fun {n} A i j => if A i j = true then 1 else -1
def Conjecture236.realMatrix : {n : ℕ} → Conjecture236.SignMatrix n → Matrix (Fin n) (Fin n) ℝ :=
fun {n} A i j => ↑(Conjecture236.integerMatrix A i j)
def Conjecture236.gram : {n : ℕ} → Conjecture236.SignMatrix n → Matrix (Fin n) (Fin n) ℝ :=
fun {n} A => Conjecture236.realMatrix A * (Conjecture236.realMatrix A).transpose
def Conjecture236.IsSignMatrix : {n : ℕ} → Matrix (Fin n) (Fin n) ℝ → Prop :=
fun {n} A => ∀ (i j : Fin n), A i j = 1 ∨ A i j = -1
def Conjecture236.D : ℕ → ℝ :=
fun n => Finset.univ.sup' ⋯ fun A => |(Conjecture236.realMatrix A).det|
def Conjecture236.IsHadamard : {n : ℕ} → Matrix (Fin n) (Fin n) ℝ → Prop :=
fun {n} H => Conjecture236.IsSignMatrix H ∧ H * H.transpose = ↑n • 1
def Conjecture236.HadamardConjecture : Prop :=
∀ (n : ℕ), 0 < n → 4 ∣ n → ∃ H, Conjecture236.IsHadamard H
def Conjecture236.normalizedMaximum : ℕ → ℝ :=
fun n => Conjecture236.D n / ↑n ^ (↑n / 2)
def Conjecture236.AllOrdersLimit : Prop :=

def Conjecture236.OriginalConjecture : Prop :=
