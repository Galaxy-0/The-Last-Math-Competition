def Conjecture2802.LatticeLaw : MeasureTheory.Measure ℝ → Prop :=
fun μ => ∃ a h, 0 < h ∧ ∀ᵐ (x : ℝ) ∂μ, ∃ k, x = a + h * ↑k

def Conjecture2802.cramerRate : MeasureTheory.Measure ℝ → ℝ → EReal :=

def Conjecture2802.RateC2On : (ℝ → EReal) → Set ℝ → Prop :=
fun I S => ∃ f, ContDiffOn ℝ 2 f S ∧ ∀ x ∈ S, I x = ↑(f x)

  ∀ (μ : MeasureTheory.Measure ℝ),
