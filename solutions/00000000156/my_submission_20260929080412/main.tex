\begin{claim}
The probability that the determinant of a random $\pm 1$ $n\times n$ matrix is prime in
absolute value is $\sim c/n$ (with an explicit formula for $c$).
\end{claim}

\begin{center}\fbox{\textbf{Verdict: FALSE.}}\end{center}

\section{The attack}

\begin{prop}\label{prop:div}
Let $A$ be an $n\times n$ matrix with all entries in $\{-1,1\}$. Then
$2^{\,n-1}\mid \det A$.
\end{prop}

\begin{proof}
Subtract the first row of $A$ from each of the rows $2,\dots,n$. This row operation does
not change the determinant. In the resulting matrix $A'$, every entry of rows
$2,\dots,n$ lies in $\{0,-2\}$. Multiplying each of those $n-1$ rows by $1/2$ multiplies
the determinant by $2^{\,n-1}$, so
\[
\det A = 2^{\,n-1}\det B
\]
for an integer matrix $B$ (its first row is the first row of $A$, and its remaining rows
have entries in $\{0,-1\}$). Hence $2^{\,n-1}\mid \det A$.
\end{proof}

\begin{cor}\label{cor:noprime}
For $n\ge 3$, no $\pm 1$ $n\times n$ matrix has prime $|\det|$.
\end{cor}

\begin{proof}
If $\det A=0$, $|\det A|$ is not prime. Otherwise, by Proposition~\ref{prop:div} and
$n\ge 3$, the integer $|\det A|$ is a positive multiple of $2^{\,n-1}\ge 4$; in
particular it is even and at least $4$. The only even prime is $2$, and $4\nmid 2$, so
$|\det A|$ is composite.
\end{proof}

\begin{cor}[Disproof]\label{cor:main}
Let $p_n$ be the probability that $|\det A|$ is prime for a uniformly random $\pm 1$
$n\times n$ matrix. Then $p_n=0$ for every $n=1$ and every $n\ge 3$ (Corollary
\ref{cor:noprime}; for $n=1$, $|\det|=1$). Since $p_n\equiv 0$ for all $n\ge 3$, we have
$p_n/(c/n)\to 0\ne 1$ for every constant $c>0$. The conjectured asymptotic $\sim c/n$
(with any positive explicit $c$) is false.
\end{cor}

\section{Boundary: exact small-$n$ behavior}

\begin{itemize}
\item \textbf{$n=1$.} $\det\in\{-1,1\}$: never prime, $p_1=0$.
\item \textbf{$n=2$.} $A=\left(\begin{smallmatrix}a&b\\ c&d\end{smallmatrix}\right)$,
$\det A=ad-bc$ with $ad,bc\in\{-1,1\}$, so $\det A\in\{-2,0,2\}$ and $|\det A|=2$ is
prime exactly when $ad\ne bc$, which happens in $8$ of the $2^4=16$ matrices:
\[
p_2=\tfrac{8}{16}=\tfrac12 .
\]
This is the \emph{only} dimension in which a prime determinant can occur
(Corollary~\ref{cor:noprime}). Matching $p_2=c/2=1/2$ would force $c=1$, but then
$c/n=1/n$ versus the true values $p_n=0$ for all $n\ge3$; the conjecture fails there and
hence asymptotically.
\item \textbf{$n=3$ (exhaustive, $2^9=512$ matrices).} All $512$ determinants are
divisible by $4$, and $\det\in\{-4,0,4\}$: no primes, $p_3=0$.
\item \textbf{$n=4$ (exhaustive, $2^{16}=65536$ matrices).} All $65536$ determinants are
divisible by $8$, and $\det\in\{0,\pm8,\pm16\}$: no primes, $p_4=0$.
\item \textbf{$n=5,6,7,8$ (randomized).} $3000$ samples each (reproduce.py, seed
$156$): $100\%$ divisible by $2^{\,n-1}$, zero prime $|\det|$.
\end{itemize}

\section{Formal verification (core Lean, no Mathlib)}

The package contains a Lean~4 (toolchain v4.33.1, core only) development:

\begin{itemize}
\item \texttt{no\_prime\_det}: the arithmetic heart — a number $m$ with $4 \mid m$
(written $m = 2\cdot(2\cdot q)$) is never prime; applied to $m = |\det A|$ for
$n \ge 3$, where $2^{n-1} \mid \det A$ gives $4 \mid |\det A|$, this yields
$p_n = 0$ for all $n \ge 3$ and refutes the $\sim c/n$ asymptotic;
\item \texttt{two\_is\_possible}: $m=2$ is prime and even, pinpointing
$n=2$ as the only feasible case;
\item exhaustive \texttt{decide} checks over the $16$ determinants of all
$2\times2$ and the $512$ determinants of all $3\times3$ $\pm1$ matrices (bit
enumeration, regenerated by \texttt{reproduce.py}): divisibility by
$2^{\,n-1}$, absence of prime $|\det|$, and the count $8$ of prime
determinants at $n=2$.
\end{itemize}

\texttt{Check.lean} runs \texttt{\#print axioms} on every theorem; all report
\emph{``does not depend on any axioms''}. There are no \texttt{sorry} statements.
