\begin{abstract}
Verdict: \textbf{FALSE}. The conjecture asserts that for every algebraic irrational
$\alpha$ the geometric means of continued-fraction partial quotients,
$G_N(\alpha)=\bigl(\prod_{j\le N} a_j\bigr)^{1/N}$, satisfy
$\liminf_{N\to\infty} G_N(\alpha)=1$ and $\limsup_{N\to\infty}G_N(\alpha)=\infty$.
The counterexample $\alpha=\sqrt2=[1;2,2,2,\dots]$ is algebraic irrational, yet all its
partial quotients after the first equal $2$, so
$G_N(\sqrt2)=2^{(N-1)/N}\to 2$. The sequence is confined to $[\sqrt2,\,2)$, whence
$\liminf G_N=\limsup G_N=2$: both assertions fail simultaneously.
\end{abstract}

\section{The conjecture}

\begin{quote}
\textbf{00000000340.} Let $G_N(\alpha)=\bigl(\prod_{j\le N}a_j\bigr)^{1/N}$, where
$a_1,a_2,\dots$ are the partial quotients of the simple continued fraction of $\alpha$.
Conjecture: for every algebraic irrational $\alpha$,
\[
\liminf_{N\to\infty}G_N(\alpha)=1
\qquad\text{and}\qquad
\limsup_{N\to\infty}G_N(\alpha)=\infty .
\]
\end{quote}

\section{Counterexample: $\alpha=\sqrt2$}

The regular continued fraction of $\sqrt2$ is
\[
\sqrt2=[1;\,2,2,2,\dots],
\]
i.e.\ $a_1=1$ and $a_j=2$ for every $j\ge 2$. This is exact, not asymptotic: writing the
standard integer CF recurrence for $\sqrt D$ ($m\leftarrow da-m$, $d\leftarrow (D-m^2)/d$,
$a\leftarrow\lfloor(\lfloor\sqrt D\rfloor+m)/d\rfloor$) with $D=2$ gives $(m,d)=(0,1)$, then the
period $(1,2)$, and one checks $p_k^2-2q_k^2=(-1)^{k+1}$ for the convergents (Pell's equation),
which forces $a_j=2$ for all $j\ge2$.

Therefore, for every $N\ge1$,
\[
\prod_{j\le N}a_j = 1\cdot 2^{N-1}=2^{N-1},
\qquad
G_N(\sqrt2)=\bigl(2^{N-1}\bigr)^{1/N}=2^{(N-1)/N}=2\cdot 2^{-1/N}.
\]

\section{Why both claims fail}

The sequence $G_N(\sqrt2)=2^{(N-1)/N}$ is strictly increasing (for $N\ge1$) and converges to
$2$. Consequently
\[
\liminf_{N\to\infty}G_N(\sqrt2)=\limsup_{N\to\infty}G_N(\sqrt2)=2,
\]
so

\begin{itemize}
\item $\limsup G_N=\infty$ \textbf{fails}: $G_N\le 2$ for every $N$ (indeed $G_N<2$);
\item $\liminf G_N=1$ \textbf{fails}: $G_N\ge\sqrt2>1$ for every $N\ge2$; more sharply,
      $G_N\ge1.99$ for every $N\ge139$.
\end{itemize}

All inequalities used above are provable in elementary integer arithmetic; e.g.\
$G_N\ge1.99$ is the integer inequality $199^N\le 2^{N-1}\cdot 100^N$, and the threshold
$N=139$ is \emph{sharp}: at $N=138$ the inequality fails ($199^{138}>2^{137}\cdot100^{138}$).
At $N=1000$: $199^{1000}\le 2^{999}\cdot100^{1000}\le 201^{1000}$, i.e.\
$1.99\le G_{1000}\le 2.01$, and in fact $G_{1000}=1.998614186\dots$

These statements are certified in machine-checked form in \texttt{lean4/Main.lean}
(core Lean~4 only, no Mathlib, and \emph{zero axioms}: every proof is by
\texttt{rfl}/\texttt{decide} and explicit $\mathbb{N}$-lemmas; \texttt{Check.lean} audits
each theorem with \texttt{\#print axioms}).

\section{Boundary of the disproof}

\begin{itemize}
\item \textbf{The failure is structural, not an $\sqrt2$ accident.} Every quadratic irrational
has an eventually periodic continued fraction, hence bounded partial quotients and bounded
$(G_N)$; moreover $G_N$ is bounded below by a constant $>1$ (the geometric mean of one period).
So the conjecture fails on the whole class of quadratic irrationals (indeed on all badly
approximable numbers).
\item \textbf{The almost-everywhere version fails too.} For Lebesgue-a.e.\ $\alpha$, Birkhoff's
ergodic theorem applied to the Gauss measure $d\mu=\frac{1}{\ln2}\frac{dx}{1+x}$ gives
$\frac1N\sum_{j\le N}\ln a_j\to\int\ln a\,d\mu=\ln K$, so $G_N(\alpha)\to K\approx2.685452001$
(Khinchin's constant). Thus $\limsup G_N=\infty$ fails for a.e.\ $\alpha$ as well; the
conjecture cannot be rescued by replacing ``algebraic'' with ``almost every''.
\item \textbf{What is not settled.} Whether some non-quadratic algebraic irrational has
unbounded partial quotients (e.g.\ for $\alpha=\sqrt[3]{2}$) is a classical open problem.
The present disproof does not touch it; it only shows that the conjecture \emph{as stated} is
false, because a single algebraic irrational ($\sqrt2$, or any quadratic irrational) refutes it.
\end{itemize}
