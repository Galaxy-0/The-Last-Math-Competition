def Conjecture141.PrimeCycles : {n : ℕ} → Equiv.Perm (Fin n) → Prop :=
fun {n} σ => ∀ k ∈ σ.partition.parts, Nat.Prime k
def Conjecture141.N : ℕ → ℕ :=
fun n => Fintype.card { σ // Conjecture141.PrimeCycles σ }
def Conjecture141.crossMatching.{u_1} : {α : Type u_1} → Equiv.Perm α → Equiv.Perm (α ⊕ α) :=
fun {α} σ =>
  { toFun := Sum.elim (fun x => Sum.inr (σ x)) fun x => Sum.inl ((Equiv.symm σ) x),
    invFun := Sum.elim (fun x => Sum.inr (σ x)) fun x => Sum.inl ((Equiv.symm σ) x), left_inv := ⋯, right_inv := ⋯ }
def Conjecture141.doubleFinEquiv : (n : ℕ) → Fin n ⊕ Fin n ≃ Fin (2 * n) :=
fun n => finSumFinEquiv.trans (finCongr ⋯)
def Conjecture141.matchingPermutation : {n : ℕ} → Equiv.Perm (Fin n) → Equiv.Perm (Fin (2 * n)) :=
fun {n} σ => (Conjecture141.doubleFinEquiv n).permCongr (Conjecture141.crossMatching σ)
def Conjecture141.matchingEmbedding : (n : ℕ) → Equiv.Perm (Fin n) ↪ { σ // Conjecture141.PrimeCycles σ } :=
fun n => { toFun := fun σ => ⟨Conjecture141.matchingPermutation σ, ⋯⟩, inj' := ⋯ }
def Conjecture141.proposedCount : ℝ → ℕ → ℝ :=
fun c n => c * ↑n ^ (-1 / 2) * Real.exp (2 * √(↑n / Real.log ↑n))
def Conjecture141.sourceConstant : ℝ :=
(4 * Real.pi) ^ (-1 / 2) * Real.exp (-1 / 2)
def Conjecture141.ConjectureClaim : Prop :=
Asymptotics.IsEquivalent Filter.atTop (fun n => ↑(Conjecture141.N n))
  (Conjecture141.proposedCount Conjecture141.sourceConstant)
