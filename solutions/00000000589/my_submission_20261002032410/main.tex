\begin{abstract}
Conjecture 00000000589 asserts that for every three-generated numerical
semigroup $S=\langle a,b,c\rangle$ the Frobenius number satisfies
$g(S)\le \sqrt{3}\,(abc)^{1/3}(1+o(1))$, with the constant $\sqrt3$ optimal.
We disprove both claims with the family
$S_n=\langle n,\,n{+}1,\,n^2{-}n{-}1\rangle$: for every $n\ge 3$,
$g(S_n)=n^2-2n-1$ exactly, so
$g(S_n)/\bigl(\sqrt3\,(abc)^{1/3}\bigr)\sim n^{2/3}/\sqrt3\to\infty$.
The bound is first violated at $n=6$ (ratio $1.2434$) and fails by an
unbounded multiplicative factor, so no constant can replace $\sqrt3$.
All claims are verified in Lean~4 with zero axioms and zero \texttt{sorry}.
\end{abstract}

\section{The conjecture}

\begin{conjecture}[00000000589]
Let $g(S)$ denote the Frobenius number of a numerical semigroup $S$. For every
three-generated semigroup $S=\langle a,b,c\rangle$,
\[
g(S)\le \sqrt{3}\,(abc)^{1/3}\,(1+o(1)),
\]
and the constant $\sqrt3$ is optimal (Erd\H{o}s--Graham type).
\end{conjecture}

\section{The attack family}

For $n\ge 3$ let
\[
S_n=\langle n,\; n+1,\; n^2-n-1\rangle,\qquad
N=n^2-2n-1,\qquad c_0=n^2-n-1 .
\]
Note that $\gcd(n,n+1)=1$, hence $S_n$ is cofinite and $g(S_n)$ is defined.
Also $c_0=g(\langle n,n+1\rangle)$: the third generator is precisely the
Frobenius number of the first two.

\begin{theorem}\label{thm:main}
For every $n\ge 3$,
\[
g(S_n)=N=n^2-2n-1 .
\]
\end{theorem}

\begin{proof}
\emph{$N$ is not representable.} Suppose
$N=xn+y(n{+}1)+z\,c_0$ with $x,y,z\ge 0$.
If $z\ge 1$, the right side is at least $c_0=n^2-n-1>N=n^2-2n-1$
(because $n>0$), impossible. So $z=0$ and, writing $u=x+y$,
\[
N = xn+y(n{+}1) = un+y .
\]
Reducing mod $n$ and using $N=(n-3)n+(n-1)$ (valid for $n\ge 3$) gives
$y\equiv n-1\pmod n$, hence $y\ge n-1$. Since $u\ge y$,
\[
un = N-y \le N-(n-1) = (n-3)n,
\]
so $u\le n-3<n-1\le y\le u$, a contradiction.

\emph{Everything larger is representable.} We show every $m\ge N+1=n^2-2n$
lies in $S_n$, by strong induction on $m$.
\begin{itemize}
\item \emph{Window.} If $m=n^2-2n+i$ with $0\le i\le n-2$, then
$m=(n-2-i)\,n+i(n{+}1)$, a combination of $n$ and $n+1$.
\item \emph{Generator point.} If $m=n^2-n-1=c_0$, use the third generator.
\item \emph{Above.} If $m\ge n^2-n$, then $m-n\ge n^2-2n$; by induction
$m-n$ is representable and one extra copy of $n$ gives $m$.
\end{itemize}
\end{proof}

\section{The ratio diverges}

\begin{corollary}\label{cor:ratio}
With $a=n$, $b=n+1$, $c=c_0$,
\[
\frac{g(S_n)}{\sqrt3\,(abc)^{1/3}}
=\frac{n^2-2n-1}{\sqrt3\,\bigl(n(n+1)(n^2-n-1)\bigr)^{1/3}}
\sim \frac{n^{2/3}}{\sqrt3}\to\infty .
\]
\end{corollary}

\begin{proof}
Since $n^2-2n-1\sim n^2$ and $n(n+1)(n^2-n-1)\sim n^4$, the quotient is
asymptotic to $n^{2/3}/\sqrt3$.
\end{proof}

\begin{corollary}
Conjecture 00000000589 is false. The inequality
$g(S)\le\sqrt3(abc)^{1/3}$ fails for every $n\ge 6$ in the family, and the
ratio is unbounded, so the constant $\sqrt3$ is not merely suboptimal: no
finite constant works for three-generated semigroups in this generality.
\end{corollary}

\section{Root-free certificates and Lean verification}

The inequality $g>\sqrt3\,(abc)^{1/3}$ for positive $g,abc$ is equivalent to
$g^6>27(abc)^2$; since $3\sqrt3<5$, the weaker rational certificate
$g^3>5\,abc$ suffices. The Lean development checks, e.g.,
\[
5\cdot 1218=6090<12167=23^3\quad(n=6),\qquad
5\cdot 159180=795900<46268279=359^3\quad(n=20),
\]
\[
5\cdot 99979900=499899500<940903909399=9799^3\quad(n=100).
\]

The Lean~4 sources (toolchain \texttt{leanprover/lean4:v4.33.1}, no Mathlib)
prove Theorem~\ref{thm:main} for \emph{arbitrary} $n\ge3$ as
\texttt{frob\_family}, instantiate it at $n=6,20,100$, and check the
certificates by \texttt{decide}. The audit \texttt{Check.lean} shows that all
31 declarations (including the hand-rolled truncated-subtraction toolkit
\texttt{my\_*}) satisfy
\begin{center}\ttfamily
\ldots does not depend on any axioms
\end{center}
i.e.\ no \texttt{sorryAx}, no \texttt{Classical.choice}, no
\texttt{Quot.sound}, no \texttt{propext}. Reproduce with
\texttt{lake build \&\& lake env lean Check.lean} inside \texttt{lean4/};
the numerics with \texttt{python3 reproduce.py}.

\section{Boundary remarks}

\begin{itemize}
\item $n=2$: $n^2-n-1=1$, so $S_2$ contains $1$; the Frobenius number is
undefined and the family starts at $n=3$.
\item $n=3,4,5$ still satisfy the conjectured bound; $n=6$ is the first
counterexample.
\item The construction shows why the conjecture cannot be repaired by any
constant: the two-generator obstruction $g(\langle a,b\rangle)\sim ab$ is
carried into the three-generated world by taking the third generator
$c=g(\langle a,b\rangle)$, which lowers the Frobenius number only marginally
(here from $n^2-n-1$ to $n^2-2n-1$).
\end{itemize}
