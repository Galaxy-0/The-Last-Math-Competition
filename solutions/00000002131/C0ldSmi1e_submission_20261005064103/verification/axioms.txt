fun n => Equiv.Perm (Fin n)
def Conjecture2131.Contains : {k n : ℕ} → Conjecture2131.Permutation k → Conjecture2131.Permutation n → Prop :=
fun {k n} p s => ∃ e, StrictMono e ∧ ∀ (i j : Fin k), s (e i) < s (e j) ↔ p i < p j
def Conjecture2131.Avoids : {k n : ℕ} → Conjecture2131.Permutation k → Conjecture2131.Permutation n → Prop :=
fun {k n} p s => ¬Conjecture2131.Contains p s

fun {k l} p q => finSumFinEquiv.symm.trans ((Equiv.sumCongr p q).trans finSumFinEquiv)

def Conjecture2131.decreasing : (n : ℕ) → Conjecture2131.Permutation n :=

def Conjecture2131.rootCount : {k : ℕ} → Conjecture2131.Permutation k → ℕ → ℝ :=
fun {k} p n => ↑(Conjecture2131.avoidanceCount p n) ^ (1 / ↑n)
def Conjecture2131.rootSequence : {k : ℕ} → Conjecture2131.Permutation k → ℕ → ℝ :=
fun {k} p n => Conjecture2131.rootCount p (n + 1)
def Conjecture2131.SW : {k : ℕ} → Conjecture2131.Permutation k → ℝ → Prop :=
fun {k} p x => Filter.Tendsto (Conjecture2131.rootSequence p) Filter.atTop (nhds x)
def Conjecture2131.UniversalMultiplicativity : Prop :=
∀ (k l : ℕ),
  0 < k →
    0 < l →
      ∀ (p : Conjecture2131.Permutation k) (q : Conjecture2131.Permutation l),
        ∃ x y z,
          Conjecture2131.SW p x ∧ Conjecture2131.SW q y ∧ Conjecture2131.SW (Conjecture2131.directSum p q) z ∧ z = x * y
def Conjecture2131.NumericalMultiplicativity : ((k : ℕ) → Conjecture2131.Permutation k → ℝ) → Prop :=
fun L =>
  ∀ (k l : ℕ),
    0 < k →
      0 < l →
        ∀ (p : Conjecture2131.Permutation k) (q : Conjecture2131.Permutation l),
          L (k + l) (Conjecture2131.directSum p q) = L k p * L l q
