#!/usr/bin/env python3
import hashlib, json, pathlib, subprocess, sys, datetime
root = pathlib.Path('/private/tmp/tlmc106-proof')
packages = pathlib.Path('/private/tmp/tlmc310-proof/.lake/packages')
bin_dir = pathlib.Path('/private/tmp/tlmc310-runtime/lean-4.19.0-darwin_aarch64/bin')
label = sys.argv[1]
def run(argv, cwd=None):
    r = subprocess.run(argv, cwd=cwd, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    return {'argv': [str(x) for x in argv], 'cwd': str(cwd) if cwd else None,
            'returncode': r.returncode, 'output': r.stdout}
manifest = json.loads((root/'lake-manifest.json').read_text())
data = {'label': label, 'utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
        'python': sys.version, 'lean': run([bin_dir/'lean', '--version']),
        'lean_githash': run([bin_dir/'lean', '--githash']),
        'config_sha256': {name: hashlib.sha256((root/name).read_bytes()).hexdigest()
          for name in ['lakefile.toml','lake-manifest.json','lean-toolchain']}, 'dependencies': []}
for pkg in manifest['packages']:
    path=packages/pkg['name']
    head=run(['git','rev-parse','HEAD'],path)
    status=run(['git','status','--porcelain','--untracked-files=no'],path)
    item={'name':pkg['name'],'expected_rev':pkg['rev'],'head':head,'tracked_status':status}
    item['matches_pin']=head['returncode']==0 and head['output'].strip()==pkg['rev']
    item['tracked_clean']=status['returncode']==0 and status['output']==''
    data['dependencies'].append(item)
data['all_nine_pinned_clean']=len(data['dependencies'])==9 and all(d['matches_pin'] and d['tracked_clean'] for d in data['dependencies'])
out=pathlib.Path('/private/tmp/tlmc106-auxiliary')/f'environment-{label}.json'
out.write_text(json.dumps(data,indent=2)+'\n')
print(out)
print('all_nine_pinned_clean =',data['all_nine_pinned_clean'])
assert data['all_nine_pinned_clean']
