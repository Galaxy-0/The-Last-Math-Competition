def Conjecture106.largestPrimeFactor : ℕ → ℕ :=
fun m => m.primeFactors.sup id
def Conjecture106.Good : ℕ → ℝ → ℕ → Prop :=
fun k c n => 2 ≤ n ∧ ∀ (i : ℕ), 1 ≤ i → i ≤ k → ↑n ^ c < ↑(Conjecture106.largestPrimeFactor (n + i))
def Conjecture106.Statement : Prop :=
∀ (k : ℕ), 2 ≤ k → ∃ c, 0 < c ∧ {n | Conjecture106.Good k c n}.Infinite
